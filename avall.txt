from pathlib import Path
CONFIG_PATH = Path.cwd() / "AVALL.cfg"

def load_cfg():
    cfg = {}
    if CONFIG_PATH.exists():
        for line in CONFIG_PATH.read_text().splitlines():
            if "=" in line:
                k, v = line.split("=", 1)
                cfg[k.strip()] = v.strip()
    return cfg

def read_cfg(key):
    return load_cfg().get(key)

def save_cfg(cfg):
    CONFIG_PATH.parent.mkdir(parents=True, exist_ok=True)
    lines = [f"{k}={v}" for k, v in cfg.items()]
    CONFIG_PATH.write_text("\n".join(lines) + "\n")

def write_cfg(key, value):
    cfg = load_cfg()          # returns {} if the file does not exist
    cfg[key] = value
    save_cfg(cfg)             # creates the file

  Reply with only JSON in this exact form:

  def grok_depose(narrative: str) -> dict:
      default = {
          "design": 0, "equipment": 0, "procedures": 0,
          "operators": 0, "supplies": 0, "environment": 0,
          "unanticipated_interaction": 0
      }
      if not isinstance(narrative, str) or len(narrative.strip()) < 50:
          return default
      try:
          resp = grok_client.chat.completions.create(
              model="grok-3",
              messages=[{"role": "user", "content": prompt_template.format(narrative=narrative[:4000])}],
              temperature=0
          )
          text = resp.choices[0].message.content.strip()
          result = json.loads(text)
          return {k: int(result.get(k, 0)) for k in default}
      except Exception:
          return default

import pandas as pd

df = pd.read_parquet("/content/drive/MyDrive/Colab Notebooks/avall_export/fact_with_grok.parquet")

pairs = [
    ("design",      "depose_design_struct",      "design_grok"),
    ("equipment",   "depose_equipment_struct",   "equipment_grok"),
    ("procedures",  "depose_procedures_struct",  "procedures_grok"),
    ("operators",   "depose_operators_struct",   "operators_grok"),
    ("supplies",    "depose_supplies_struct",    "supplies_grok"),
    ("environment", "depose_environment_struct", "environment_grok"),
]

# only the rows Grok tagged
mask = df[[g for _, _, g in pairs]].notna().any(axis=1)
sub = df.loc[mask].copy()

rows = []
for name, s_col, g_col in pairs:
    s = sub[s_col].fillna(0).astype(int)
    g = sub[g_col].fillna(0).astype(int)

    both     = ((s == 1) & (g == 1)).sum()
    struct_only = ((s == 1) & (g == 0)).sum()
    grok_only   = ((s == 0) & (g == 1)).sum()
    neither     = ((s == 0) & (g == 0)).sum()

    rows.append({
        "category": name,
        "struct_only": struct_only,
        "grok_only": grok_only,
        "both": both,
        "neither": neither,
        "grok_adds": grok_only,          # pure enrichment
        "total_with_signal": both + struct_only + grok_only,
    })

enrich = pd.DataFrame(rows).set_index("category")
print(enrich.to_string())
print()
print(f"Rows in comparison: {mask.sum():,}")
